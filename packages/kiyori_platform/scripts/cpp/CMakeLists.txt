cmake_minimum_required(VERSION 3.24)
project(kiyori_scripts LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_LIBC OFF CACHE BOOL "" FORCE)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# v0.17.0 resolves to 6d46d07d04041b40f4f49eaa7fdebe44c314c699; hash pins tag archive bytes.
set(qjs_archive https://codeload.github.com/quickjs-ng/quickjs/tar.gz/refs/tags/v0.17.0)
if(KIYORI_SCRIPT_ARCHIVE)
  file(TO_CMAKE_PATH "${KIYORI_SCRIPT_ARCHIVE}" qjs_archive)
endif()
FetchContent_Declare(quickjs URL "${qjs_archive}" URL_HASH SHA256=559BC4C420475E55C7AB4510ADBC562F55D7524D75E8E89D79CE4BB02F5687D9 DOWNLOAD_EXTRACT_TIMESTAMP TRUE EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(quickjs)
add_library(kiyori_scripts STATIC scripts.cpp)
target_link_libraries(kiyori_scripts PRIVATE qjs)
if(MSVC)
  target_compile_options(kiyori_scripts PRIVATE /utf-8)
endif()
install(TARGETS kiyori_scripts qjs ARCHIVE DESTINATION lib COMPONENT kiyori)
install(FILES "${quickjs_SOURCE_DIR}/LICENSE" DESTINATION licenses/QuickJS-NG COMPONENT kiyori)
