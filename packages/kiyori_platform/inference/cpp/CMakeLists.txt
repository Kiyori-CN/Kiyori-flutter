cmake_minimum_required(VERSION 3.24)
project(kiyori_inference LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(mnn_archive https://codeload.github.com/alibaba/MNN/tar.gz/ea44a3ebd5dd6348eea501047b17c43aa3ecccb6)
set(llama_archive https://codeload.github.com/ggml-org/llama.cpp/tar.gz/885c5bbe8e04dc78db25beb911a2715312ad7b54)
if(KIYORI_NATIVE_ARCHIVES)
  file(TO_CMAKE_PATH "${KIYORI_NATIVE_ARCHIVES}" KIYORI_NATIVE_ARCHIVES)
  set(mnn_archive "${KIYORI_NATIVE_ARCHIVES}/mnn.tar.gz")
  set(llama_archive "${KIYORI_NATIVE_ARCHIVES}/llama.tar.gz")
endif()
FetchContent_Declare(mnn URL "${mnn_archive}" URL_HASH SHA256=C508AF915C4C07A1885E908F3FB365B2BFDE425777BF0266F9ECD71F734A185A DOWNLOAD_EXTRACT_TIMESTAMP TRUE EXCLUDE_FROM_ALL)
FetchContent_Declare(llama URL "${llama_archive}" URL_HASH SHA256=C3E9A3BC13A6FFB3E640F5EC8D5BD85CCBBC2172273D144DE1BA74F1179766A5 DOWNLOAD_EXTRACT_TIMESTAMP TRUE EXCLUDE_FROM_ALL)
foreach(flag MNN_BUILD_LLM MNN_LOW_MEMORY MNN_SUPPORT_TRANSFORMER_FUSE MNN_CPU_WEIGHT_DEQUANT_GEMM)
  set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
foreach(flag MNN_BUILD_SHARED_LIBS MNN_SEP_BUILD MNN_BUILD_BENCHMARK MNN_BUILD_TEST MNN_BUILD_TOOLS MNN_BUILD_QUANTOOLS MNN_EVALUATION MNN_BUILD_CONVERTER MNN_BUILD_TRAIN MNN_LLM_BUILD_DEMO MNN_LLM_BUILD_TEST MNN_BUILD_DEMO MNN_OPENMP MNN_VULKAN MNN_OPENCL MNN_OPENGL MNN_METAL MNN_KLEIDIAI MNN_BUILD_LLM_OMNI BUILD_MLS)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_CURL GGML_NATIVE GGML_OPENMP GGML_VULKAN GGML_CUDA GGML_METAL)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLM_SUPPORT_HTTP_RESOURCE OFF CACHE BOOL "" FORCE)
if(ANDROID)
  # 由Cargo安装静态库；关闭上游JNI目录复制分支，OBJECT llm没有POST_BUILD阶段。
  set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
endif()
FetchContent_MakeAvailable(mnn llama)
add_library(kiyori_inference STATIC inference.cpp)
target_include_directories(kiyori_inference PRIVATE "${mnn_SOURCE_DIR}/include" "${mnn_SOURCE_DIR}/transformers/llm/engine/include" "${mnn_SOURCE_DIR}/transformers/llm/engine/src" "${mnn_SOURCE_DIR}/3rd_party" "${llama_SOURCE_DIR}/include" "${llama_SOURCE_DIR}/ggml/include" "${llama_SOURCE_DIR}/vendor")
target_link_libraries(kiyori_inference PRIVATE MNN llama)
if(MSVC)
  target_compile_options(kiyori_inference PRIVATE /utf-8)
endif()
install(TARGETS kiyori_inference MNN llama ggml ggml-cpu ggml-base ARCHIVE DESTINATION lib COMPONENT kiyori)
install(FILES "${mnn_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses/MNN COMPONENT kiyori)
install(FILES "${llama_SOURCE_DIR}/LICENSE" DESTINATION licenses/llama.cpp COMPONENT kiyori)
